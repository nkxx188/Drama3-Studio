cmake_minimum_required(VERSION 3.20)
project(drama_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(drama_native SHARED
    src/runtime.cpp src/fp8_reference.cpp src/fp8_avx2.cpp src/fp8_avx512.cpp)
target_include_directories(drama_native PRIVATE include src)
target_compile_definitions(drama_native PRIVATE DRAMA_NATIVE_BUILD)
if(MSVC)
    # ISA-specific instructions live in separate files behind runtime checks.
    target_compile_options(drama_native PRIVATE /O2 /W4 /EHsc /fp:precise)
    set_source_files_properties(src/fp8_avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    set_source_files_properties(src/fp8_avx512.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX512")
    set_property(TARGET drama_native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
else()
    target_compile_options(drama_native PRIVATE -O3 -Wall -Wextra -ffp-contract=off)
    set_source_files_properties(src/fp8_avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
    set_source_files_properties(src/fp8_avx512.cpp PROPERTIES COMPILE_OPTIONS "-mavx512f;-mavx512bw;-mfma")
endif()
